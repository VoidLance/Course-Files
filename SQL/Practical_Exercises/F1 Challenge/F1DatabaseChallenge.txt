Results of the thirty-sixth challenge:

I'm going to only record the queries I write from here on, it seems more sensible but there's not enough information in the practical activity instructions to know whether that's what I'm supposed to do.

SELECT d.GivenName || ' ' || d.FamilyName AS FullName, r.ConstructorName
FROM results r
INNER JOIN drivers d ON r.DriverID = d.driverId
WHERE r.Season = 2000;

Query for the thirty-seventh challenge:


SELECT c.Name, r.GivenName || ' ' || r.FamilyName AS FullName
FROM constructors c
LEFT JOIN results r ON c.Name = r.ConstructorName
GROUP BY c.Name, FullName
ORDER BY FullName IS NULL, c.Name, r.FamilyName

Query for the thirty-eighth challenge:

SELECT r.*, d.driverId
FROM drivers d
RIGHT JOIN results r ON d.driverId = r.DriverID

Query for the thirty-ninth challenge:

SELECT d.DriverID, d.GivenName, d.FamilyName, r.Season, r.Round, r.Position, r.Points
FROM drivers d
LEFT JOIN results r ON d.DriverID = r.DriverID;

Query for the fourtieth challenge:

SELECT d.GivenName, d.FamilyName, c.Name AS ConstructorName, COALESCE(SUM(r.Points), 0) AS TotalPoints
FROM drivers d
LEFT JOIN results r ON d.DriverID = r.DriverID AND r.Season = 2000
LEFT JOIN constructors c ON r.ConstructorName = c.Name
GROUP BY d.DriverID, d.GivenName, d.FamilyName, c.Name;

Query for the fourty-first challenge:

SELECT DriverID, SUM(Points) AS TotalPoints
FROM results
WHERE Season = 2000
GROUP BY DriverID
HAVING SUM(Points) > (
    SELECT MIN(TotalPoints)
    FROM (
        SELECT SUM(Points) AS TotalPoints
        FROM results
        WHERE Season = 2000
        GROUP BY DriverID
    ) AS MinPoints
);

Query for the fourty-second challenge:

SELECT GivenName,
       FamilyName,
       (SELECT MAX(Points)
        FROM results
        GROUP BY DriverID
        HAVING DriverID = r.DriverID) AS MaxPoints
FROM results AS r
GROUP BY DriverID, GivenName, FamilyName;

Query for the fourty-third challenge:


SELECT ConstructorName,
       GivenName,
       FamilyName, Points
FROM results AS r1
WHERE Points > (SELECT AVG(Points) AS AvgPoints
                FROM results AS r2
                WHERE r1.ConstructorName = r2.ConstructorName)
ORDER BY ConstructorName;

Query for the fourty-fourth challenge:


WITH TotalPoints AS (
    SELECT DriverID,
           SUM(Points) AS TotalPoints
    FROM results
    WHERE Season = '2000'
    GROUP BY DriverID
)
SELECT AVG(TotalPoints) AS AveragePointsPerDriver
FROM TotalPoints;

Query for the fourty-fifth challenge:

WITH TopFinishes AS (
    SELECT DriverID,
           COUNT(*) AS RacesFinishedInTop3,
           AVG(Points) AS AveragePoints
    FROM results
    WHERE Season = '2000' AND Position <= 3
    GROUP BY DriverID
)
SELECT DriverID,
       RacesFinishedInTop3,
       AveragePoints
FROM TopFinishes
WHERE RacesFinishedInTop3 > 5;
